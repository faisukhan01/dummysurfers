    32→    32→/** Unified z-sorted draw item. */
    33→    33→class DrawItem {
    34→    34→    var z = 0f
    35→    35→    var type = 0 // 0 deco,1 train,2 obstacle,3 coin,4 powerup,5 player,6 chaser
    36→    36→    var deco: Deco? = null
    37→    37→    var train: Train? = null
    38→    38→    var obstacle: Obstacle? = null
    39→    39→    var coin: Coin? = null
    40→    40→    var powerup: PowerUpPickup? = null
    41→    41→}
    42→    42→
    43→    43→/**
    44→    44→ * Renders every gameplay entity with the shared pseudo-3D projection,
    45→    45→ * z-sorted back-to-front, interleaving ShapeRenderer and SpriteBatch
    46→    46→ * efficiently by switching only when the renderer type changes.
    47→    47→ */
    48→    48→class EntityRenderer(
    49→    49→    private val proj: Projection,
    50→    50→    private val batch: SpriteBatch,
    51→    51→    private val sr: ShapeRenderer,
    52→    52→    private val fontLarge: BitmapFont,
    53→    53→    private val fontSmall: BitmapFont
    54→    54→) {
    55→    55→    private val tmpC = Color()
    56→    56→    private val tmpC2 = Color()
    57→    57→    private val layout = GlyphLayout()
    58→    58→    private val items = Array(700) { DrawItem() }
    59→    59→    private var itemCount = 0
    60→    60→
    61→    61→    private fun item(): DrawItem {
    62→    62→        val it = items[itemCount % items.size]
    63→    63→        itemCount++
    64→    64→        return it
    65→    65→    }

    67→    67→    // ── Main entry ─────────────────────────────────────────────────────
    68→    68→    fun render(
    69→    69→        world: com.dummysurfers.core.world.WorldGenerator,
    70→    70→        spawner: com.dummysurfers.core.systems.Spawner,
    71→    71→        player: Player,
    72→    72→        chaser: Chaser,
    73→    73→        character: CharacterDef,
    74→    74→        shakeX: Float,
    75→    75→        shakeY: Float,
    76→    76→        particlesFx: Particles,
    77→    77→        invulnBlink: Boolean,
    78→    78→        shieldOn: Boolean,
    79→    79→        boostOn: Boolean
    80→    80→    ) {
    81→    81→        itemCount = 0
    82→    82→        for (d in world.decos) {
    83→    83→            if (d.z > GameConfig.VIEW_DISTANCE + 6f || d.z < -14f) continue
    84→    84→            val it = item(); it.z = d.z; it.type = 0; it.deco = d
    85→    85→        }
    86→    86→        for (t in spawner.trains) { val it = item(); it.z = t.z; it.type = 1; it.train = t }
    87→    87→        for (o in spawner.obstacles) { val it = item(); it.z = o.z; it.type = 2; it.obstacle = o }
    88→    88→        for (c in spawner.coins) { if (!c.collected && c.z < GameConfig.VIEW_DISTANCE) { val it = item(); it.z = c.z; it.type = 3; it.coin = c } }
    89→    89→        for (p in spawner.powerups) { if (!p.taken) { val it = item(); it.z = p.z; it.type = 4; it.powerup = p } }
    90→    90→        if (chaser.active) { val it = item(); it.z = GameConfig.CHASER_Z; it.type = 6 }
    91→    91→        { val it = item(); it.z = 0f; it.type = 5 }
    92→    92→
    93→    93→        // far → near
    94→    94→        val order = (0 until itemCount).sortedByDescending { items[it].z }
    95→    95→        for (idx in order) {
    96→    96→            val it = items[idx]
    97→    97→            val wantBatch = it.type == 3 || it.type == 4
    98→    98→            setMode(wantBatch)
    99→    99→            when (it.type) {
   100→   100→                0 -> DecoRenderer.draw(it.deco!!, sr, proj, shakeX, shakeY)
   101→   101→                1 -> train(it.train!!, shakeX, shakeY)
   102→   102→                2 -> obstacle(it.obstacle!!, shakeX, shakeY)
   103→   103→                3 -> coin(it.coin!!, shakeX, shakeY)
   104→   104→                4 -> powerup(it.powerup!!, shakeX, shakeY)
   105→   105→                5 -> this.player(player, character, shakeX, shakeY, invulnBlink, shieldOn, boostOn)
   106→   106→                6 -> this.chaser(chaser, shakeX, shakeY)
   107→   107→            }
   108→   108→        }
   109→   109→        // particle shapes (ShapeRenderer), then floating texts (SpriteBatch)
   110→   110→        setMode(false)
   111→   111→        particlesFx.render(sr)
   112→   112→        setMode(true)
   113→   113→        drawFloatingTexts(particlesFx, shakeX, shakeY)
   114→   114→        sr.end()
   115→   115→        batch.end()
   116→   116→        began = false
   117→   117→    }

   119→   119→    private var modeBatch = false
   120→   120→    private var began = false
   121→   121→
   122→   122→    private fun setMode(batchMode: Boolean) {
   123→   123→        if (batchMode == modeBatch && began) return
   124→   124→        if (began) {
   125→   125→            if (modeBatch) batch.end() else sr.end()
   126→   126→        }
   127→   127→        modeBatch = batchMode
   128→   128→        if (modeBatch) {
   129→   129→            batch.begin()
   130→   130→        } else {
   131→   131→            sr.begin(ShapeRenderer.ShapeType.Filled)
   132→   132→        }
   133→   133→        began = true
   134→   134→    }

   136→   136→    fun fogged(out: Color, base: Color, z: Float): Color {
   137→   137→        val f = proj.fog(z) * 0.85f
   138→   138→        out.set(base)
   139→   139→        out.r += (Palette.FOG.r - out.r) * f
   140→   140→        out.g += (Palette.FOG.g - out.g) * f
   141→   141→        out.b += (Palette.FOG.b - out.b) * f
   142→   142→        return out
   143→   143→    }

   145→   145→    // ── 3D box helper (matches WorldRenderer.box3D) ────────────────────
   146→   146→    private fun box3D(
   147→   147→        wx: Float, w: Float, h: Float, zFront: Float, zBack: Float,
   148→   148→        front: Color, side: Color, top: Color, sx: Float, sy: Float
   149→   149→    ) {
   150→   150→        val sF = proj.scale(zFront)
   151→   151→        val sB = proj.scale(zBack)
   152→   152→        val xlF = proj.screenX(wx - w / 2f, zFront) + sx
   153→   153→        val xrF = proj.screenX(wx + w / 2f, zFront) + sx
   154→   154→        val ybF = proj.groundY(zFront) + sy
   155→   155→        val ytF = ybF - h * proj.ppu * sF
   156→   156→        val xlB = proj.screenX(wx - w / 2f, zBack) + sx
   157→   157→        val xrB = proj.screenX(wx + w / 2f, zBack) + sx
   158→   158→        val ybB = proj.groundY(zBack) + sy
   159→   159→        val ytB = ybB - h * proj.ppu * sB
   160→   160→
   161→   161→        sr.setColor(top)
   162→   162→        sr.triangle(xlF, ytF, xrF, ytF, xrB, ytB)
   163→   163→        sr.triangle(xlF, ytF, xrB, ytB, xlB, ytB)
   164→   164→        val camWx = proj.camX
   165→   165→        if (camWx < wx - w / 2f) {
   166→   166→            sr.setColor(side)
   167→   167→            sr.triangle(xlF, ytF, xlB, ytB, xlB, ybB)
   168→   168→            sr.triangle(xlF, ytF, xlB, ybB, xlF, ybF)
   169→   169→        } else if (camWx > wx + w / 2f) {
   170→   170→            sr.setColor(side)
   171→   171→            sr.triangle(xrF, ytF, xrB, ytB, xrB, ybB)
   172→   172→            sr.triangle(xrF, ytF, xrB, ybB, xrF, ybF)
   173→   173→        }
   174→   174→        sr.setColor(front)
   175→   175→        sr.rect(xlF, ytF, xrF - xlF, ybF - ytF)
   176→   176→    }

   179→   179→    private fun train(t: Train, sx: Float, sy: Float) {
   180→   180→        val livery = Palette.TRAIN_LIVERIES[t.livery % Palette.TRAIN_LIVERIES.size]
   181→   181→        val base = tmpC2.set(Color(livery[0]))
   182→   182→        val carLen = GameConfig.TRAIN_CAR_LENGTH
   183→   183→        val w = GameConfig.TRAIN_WIDTH
   184→   184→        val h = GameConfig.TRAIN_HEIGHT
   185→   185→
   186→   186→        for (car in t.cars - 1 downTo 0) {
   187→   187→            val zF = t.z - car * carLen
   188→   188→            val zB = zF - carLen + 0.55f // coupling gap
   189→   189→            if (zB > GameConfig.VIEW_DISTANCE + 4f || zF < -10f) continue
   190→   190→            val zFrontC = zF.coerceAtLeast(-9f)
   191→   191→            fogged(tmpC, base, zF)
   192→   192→            val front = Color(tmpC)
   193→   193→            val sideC = tmpC.cpy().mul(0.68f)
   194→   194→            val topC = tmpC.cpy().mul(1.18f)
   195→   195→
   196→   196→            // undercarriage shadow
   197→   197→            val sF = proj.scale(zFrontC)
   198→   198→            sr.setColor(0f, 0f, 0f, 0.3f)
   199→   199→            val shW = w * proj.ppu * sF * 0.6f
   200→   200→            for (lane in t.lanes) {
   201→   201→                val lx = proj.screenX(lane * GameConfig.LANE_WIDTH.toFloat(), zFrontC) + sx
   202→   202→                sr.rect(lx - shW / 2, proj.groundY(zFrontC) + sy - 2f, shW, 5f * sF + 2f)
   203→   203→            }
   204→   204→
   205→   205→            for (lane in t.lanes) {
   206→   206→                val wx = lane * GameConfig.LANE_WIDTH.toFloat()
   207→   207→                box3D(wx, w, h, zFrontC, zB.coerceAtLeast(-9f), front, sideC, topC, sx, sy)
   208→   208→                trainDetails(t, wx, zFrontC, zB, w, h, livery, sx, sy)
   209→   209→            }
   210→   210→        }
   211→   211→    }

   213→   213→    private fun trainDetails(t: Train, wx: Float, zF: Float, zB: Float, w: Float, h: Float, livery: IntArray, sx: Float, sy: Float) {
   214→   214→        if (zF > GameConfig.VIEW_DISTANCE * 0.92f || zF < -8f) return
   215→   215→        val s = proj.scale(zF)
   216→   216→        val xl = proj.screenX(wx - w / 2f, zF) + sx
   217→   217→        val xr = proj.screenX(wx + w / 2f, zF) + sx
   218→   218→        val yb = proj.groundY(zF) + sy
   219→   219→        val yt = yb - h * proj.ppu * s
   220→   220→
   221→   221→        // stripe along the bottom
   222→   222→        sr.setColor(fogged(tmpC, Color(livery[1]), zF))
   223→   223→        sr.rect(xl, yb - h * 0.22f * proj.ppu * s, xr - xl, h * 0.1f * proj.ppu * s)
   224→   224→
   225→   225→        // windshield / front face details only on front-facing car (zF close to t.z)
   226→   226→        val isFront = abs(zF - t.z) < 0.1f
   227→   227→        if (isFront) {
   228→   228→            // windshield
   229→   229→            sr.setColor(fogged(tmpC, Color(0x1f2a30ff.toInt()), zF))
   230→   230→            sr.rect(xl + (xr - xl) * 0.15f, yb - h * 0.82f * proj.ppu * s, (xr - xl) * 0.7f, h * 0.3f * proj.ppu * s)
   231→   231→            // headlights
   232→   232→            val glowCol = if (t.kind == 2) 1f else 0.75f
   233→   233→            sr.setColor(fogged(tmpC, Color(1f, 0.95f, 0.7f, 1f), zF))
   234→   234→            sr.circle(xl + (xr - xl) * 0.16f, yb - h * 0.3f * proj.ppu * s, 4.5f * s)
   235→   235→            sr.circle(xl + (xr - xl) * 0.84f, yb - h * 0.3f * proj.ppu * s, 4.5f * s)
   236→   236→            if (t.kind == 2) {
   237→   237→                // approaching: headlight beam glow
   238→   238→                sr.setColor(1f, 0.95f, 0.75f, 0.16f)
   239→   239→                sr.circle(xl + (xr - xl) * 0.5f, yb - h * 0.4f * proj.ppu * s, (xr - xl) * 0.75f)
   240→   240→            }
   241→   241→            // number plate
   242→   242→            sr.setColor(fogged(tmpC, Color(0xfff2dcff.toInt()), zF))
   243→   243→            sr.rect(xl + (xr - xl) * 0.42f, yb - h * 0.5f * proj.ppu * s, (xr - xl) * 0.16f, h * 0.09f * proj.ppu * s)
   244→   244→        } else {
   245→   245→            // side windows (visible on side face)
   246→   246→            val camWx = proj.camX
   247→   247→            val leftSide = camWx < wx - w / 2f
   248→   248→            val sxE = proj.screenX(if (leftSide) wx - w / 2f else wx + w / 2f, zF) + sx
   249→   249→            val sxE2 = proj.screenX(if (leftSide) wx - w / 2f else wx + w / 2f, zB) + sx
   250→   250→            val ybE = proj.groundY(zF) + sy
   251→   251→            val ytE = ybE - h * proj.ppu * s
   252→   252→            sr.setColor(fogged(tmpC, Color(0x24343bff.toInt()), zF))
   253→   253→            val winN = 4
   254→   254→            for (i in 0 until winN) {
   255→   255→                val ft0 = (i + 0.15f) / winN
   256→   256→                val ft1 = (i + 0.85f) / winN
   257→   257→                val wx0 = sxE + (sxE2 - sxE) * ft0
   258→   258→                val wx1 = sxE + (sxE2 - sxE) * ft1
   259→   259→                val wy0 = ybE - h * 0.78f * proj.ppu * s + (proj.groundY(zB) + sy - h * proj.ppu * proj.scale(zB) - (ybE - h * 0.78f * proj.ppu * s)) * ft0
   260→   260→                sr.rect(wx0.coerceAtLeast(wx1), wy0, abs(wx1 - wx0), h * 0.24f * proj.ppu * s)
   261→   261→            }
   262→   262→            // door
   263→   263→            sr.setColor(fogged(tmpC, Color(livery[1]).mul(0.9f), zF))
   264→   264→            val dx = sxE + (sxE2 - sxE) * 0.02f
   265→   265→            sr.rect(dx.coerceAtLeast(sxE2 * 0.99f + sxE * 0.01f), ybE - h * 0.62f * proj.ppu * s, abs(sxE2 - sxE) * 0.12f, h * 0.5f * proj.ppu * s)
   266→   266→        }
   267→   267→
   268→   268→        // graffiti tag (original art): colored arcs seeded per train
   269→   269→        if (t.seed % 3 == 0 && !isFront) {
   270→   270→            val rng = Random(t.seed + (zF * 10).toInt())
   271→   271→            val cols = arrayOf(Color(0xffd24aff.toInt()), Color(0x2dd4bfff.toInt()), Color(0xe2493bff.toInt()))
   272→   272→            sr.setColor(fogged(tmpC, cols[rng.nextInt(cols.size)], zF))
   273→   273→            val gy = yb - h * (0.3f + rng.nextFloat() * 0.2f) * proj.ppu * s
   274→   274→            sr.rect(xl + (xr - xl) * 0.1f, gy, (xr - xl) * (0.2f + rng.nextFloat() * 0.3f), 4f * s + 1f)
   275→   275→        }
   276→   276→    }

   279→   279→    private fun obstacle(o: Obstacle, sx: Float, sy: Float) {
   280→   280→        val s = proj.scale(o.z)
   281→   281→        if (o.z > GameConfig.VIEW_DISTANCE || o.z < -8f) return
   282→   282→        when (o.kind) {
   283→   283→            ObstacleKind.LOW_BARRIER -> lowBarrier(o, sx, sy, s)
   284→   284→            ObstacleKind.HIGH_BARRIER -> highBarrier(o, sx, sy, s)
   285→   285→            ObstacleKind.GATE -> gate(sx, sy, s)
   286→   286→            ObstacleKind.FENCE_FULL -> fenceFull(sx, sy, s)
   287→   287→            ObstacleKind.BLOCKADE -> blockade(o, sx, sy, s)
   288→   288→        }
   289→   289→    }

   291→   291→    private fun stripeBoard(xl: Float, xr: Float, y: Float, hpx: Float, variant: Int) {
   292→   292→        // orange/white diagonal hazard stripes
   293→   293→        val n = 6
   294→   294→        val w = xr - xl
   295→   295→        for (i in 0 until n) {
   296→   296→            val segW = w / n
   297→   297→            val even = (i + variant) % 2 == 0
   298→   298→            sr.setColor(if (even) Color(0xf28c3cff.toInt()) else Color(0xfff2dcff.toInt()))
   299→   299→            sr.rect(xl + i * segW, y, segW + 0.5f, hpx)
   300→   300→        }
   301→   301→    }

   303→   303→    private fun lowBarrier(o: Obstacle, sx: Float, sy: Float, s: Float) {
   304→   304→        val wx = o.lane * GameConfig.LANE_WIDTH.toFloat()
   305→   305→        val xl = proj.screenX(wx - 1.0f, o.z) + sx
   306→   306→        val xr = proj.screenX(wx + 1.0f, o.z) + sx
   307→   307→        val yb = proj.groundY(o.z) + sy
   308→   308→        val hpx = 0.95f * proj.ppu * s
   309→   309→        // legs
   310→   310→        sr.setColor(fogged(tmpC, Color(0x555049ff.toInt()), o.z))
   311→   311→        sr.rect(xl + (xr - xl) * 0.08f, yb - hpx, 5f * s, hpx * 0.55f)
   312→   312→        sr.rect(xr - (xr - xl) * 0.08f - 5f * s, yb - hpx, 5f * s, hpx * 0.55f)
   313→   313→        // striped board
   314→   314→        stripeBoard(xl, xr, yb - hpx, hpx * 0.45f, o.variant)
   315→   315→        sr.setColor(fogged(tmpC, Color(0x2b2b2bff.toInt()), o.z))
   316→   316→        sr.rect(xl, yb - hpx, xr - xl, 2.2f * s)
   317→   317→    }

   319→   319→    private fun highBarrier(o: Obstacle, sx: Float, sy: Float, s: Float) {
   320→   320→        val wx = o.lane * GameConfig.LANE_WIDTH.toFloat()
   321→   321→        val xl = proj.screenX(wx - 1.1f, o.z) + sx
   322→   322→        val xr = proj.screenX(wx + 1.1f, o.z) + sx
   323→   323→        val yb = proj.groundY(o.z) + sy
   324→   324→        // supports from ground to 1.15u (slide gap below)
   325→   325→        sr.setColor(fogged(tmpC, Color(0x555049ff.toInt()), o.z))
   326→   326→        sr.rect(xl, yb - 1.2f * proj.ppu * s, 6f * s, 1.2f * proj.ppu * s)
   327→   327→        sr.rect(xr - 6f * s, yb - 1.2f * proj.ppu * s, 6f * s, 1.2f * proj.ppu * s)
   328→   328→        // overhead signboard 1.2u..2.2u — teal = slide action (cool color)
   329→   329→        val boardY = yb - 2.3f * proj.ppu * s
   330→   330→        val boardH = 1.05f * proj.ppu * s
   331→   331→        sr.setColor(fogged(tmpC, Color(0x2fa08bff.toInt()), o.z))
   332→   332→        sr.rect(xl, boardY, xr - xl, boardH)
   333→   333→        sr.setColor(fogged(tmpC, Color(0x8ff2e2ff.toInt()), o.z))
   334→   334→        // down-arrow marks (slide cue)
   335→   335→        val cx = (xl + xr) / 2
   336→   336→        val n = 3
   337→   337→        for (i in 0 until n) {
   338→   338→            val ax = xl + (xr - xl) * (0.3f + i * 0.2f)
   339→   339→            sr.rect(ax - 2.5f * s, boardY + boardH * 0.3f, 5f * s, boardH * 0.4f)
   340→   340→            sr.rect(ax - 6f * s, boardY + boardH * 0.22f, 12f * s, 3.5f * s)
   341→   341→        }
   342→   342→        sr.setColor(0f, 0f, 0f, 0.25f)
   343→   343→        sr.rect(xl, boardY, xr - xl, 2.5f * s)
   344→   344→    }

   446→   446→    private fun player(p: Player, ch: CharacterDef, sx: Float, sy: Float, blink: Boolean, shieldOn: Boolean, boostOn: Boolean) {
   447→   447→        val s = 1f
   448→   448→        val x = proj.screenX(p.x, 0f) + sx

   458→   458→
   459→   459→        val bodyH = u * GameConfig.PLAYER_HEIGHT
   460→   460→        val squash = 1f - p.squash * 0.16f
   461→   461→        val isJump = p.state == PlayerState.JUMPING
   462→   462→        val isSlide = p.state == PlayerState.SLIDING
   463→   463→        val isDead = p.state == PlayerState.DEAD

   476→   476→        val rotRad = rotate * PI.toFloat() / 180f

   478→   478→            val dx = px - ox; val dy = py - oy
   479→   479→            return (ox + dx * cos(rotRad) - dy * sin(rotRad)) to (oy + dx * sin(rotRad) + dy * cos(rotRad))
