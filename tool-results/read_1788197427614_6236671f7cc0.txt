    32→/** Unified z-sorted draw item. */
    33→class DrawItem {
    34→    var z = 0f
    35→    var type = 0 // 0 deco,1 train,2 obstacle,3 coin,4 powerup,5 player,6 chaser
    36→    var deco: Deco? = null
    37→    var train: Train? = null
    38→    var obstacle: Obstacle? = null
    39→    var coin: Coin? = null
    40→    var powerup: PowerUpPickup? = null
    41→}
    42→
    43→/**
    44→ * Renders every gameplay entity with the shared pseudo-3D projection,
    45→ * z-sorted back-to-front, interleaving ShapeRenderer and SpriteBatch
    46→ * efficiently by switching only when the renderer type changes.
    47→ */
    48→class EntityRenderer(
    49→    private val proj: Projection,
    50→    private val batch: SpriteBatch,
    51→    private val sr: ShapeRenderer,
    52→    private val fontLarge: BitmapFont,
    53→    private val fontSmall: BitmapFont
    54→) {
    55→    private val tmpC = Color()
    56→    private val tmpC2 = Color()
    57→    private val layout = GlyphLayout()
    58→    private val items = Array(700) { DrawItem() }
    59→    private var itemCount = 0
    60→
    61→    private fun item(): DrawItem {
    62→        val it = items[itemCount % items.size]
    63→        itemCount++
    64→        return it
    65→    }

    67→    // ── Main entry ─────────────────────────────────────────────────────
    68→    fun render(
    69→        world: com.dummysurfers.core.world.WorldGenerator,
    70→        spawner: com.dummysurfers.core.systems.Spawner,
    71→        player: Player,
    72→        chaser: Chaser,
    73→        character: CharacterDef,
    74→        shakeX: Float,
    75→        shakeY: Float,
    76→        particlesFx: Particles,
    77→        invulnBlink: Boolean,
    78→        shieldOn: Boolean,
    79→        boostOn: Boolean
    80→    ) {
    81→        itemCount = 0
    82→        for (d in world.decos) {
    83→            if (d.z > GameConfig.VIEW_DISTANCE + 6f || d.z < -14f) continue
    84→            val it = item(); it.z = d.z; it.type = 0; it.deco = d
    85→        }
    86→        for (t in spawner.trains) { val it = item(); it.z = t.z; it.type = 1; it.train = t }
    87→        for (o in spawner.obstacles) { val it = item(); it.z = o.z; it.type = 2; it.obstacle = o }
    88→        for (c in spawner.coins) { if (!c.collected && c.z < GameConfig.VIEW_DISTANCE) { val it = item(); it.z = c.z; it.type = 3; it.coin = c } }
    89→        for (p in spawner.powerups) { if (!p.taken) { val it = item(); it.z = p.z; it.type = 4; it.powerup = p } }
    90→        if (chaser.active) { val it = item(); it.z = GameConfig.CHASER_Z; it.type = 6 }
    91→        { val it = item(); it.z = 0f; it.type = 5 }
    92→
    93→        // far → near
    94→        val order = (0 until itemCount).sortedByDescending { items[it].z }
    95→        for (idx in order) {
    96→            val it = items[idx]
    97→            val wantBatch = it.type == 3 || it.type == 4
    98→            setMode(wantBatch)
    99→            when (it.type) {
   100→                0 -> DecoRenderer.draw(it.deco!!, sr, proj, shakeX, shakeY)
   101→                1 -> train(it.train!!, shakeX, shakeY)
   102→                2 -> obstacle(it.obstacle!!, shakeX, shakeY)
   103→                3 -> coin(it.coin!!, shakeX, shakeY)
   104→                4 -> powerup(it.powerup!!, shakeX, shakeY)
   105→                5 -> this.player(player, character, shakeX, shakeY, invulnBlink, shieldOn, boostOn)
   106→                6 -> this.chaser(chaser, shakeX, shakeY)
   107→            }
   108→        }
   109→        // particle shapes (ShapeRenderer), then floating texts (SpriteBatch)
   110→        setMode(false)
   111→        particlesFx.render(sr)
   112→        setMode(true)
   113→        drawFloatingTexts(particlesFx, shakeX, shakeY)
   114→        sr.end()
   115→        batch.end()
   116→        began = false
   117→    }
   118→
   119→    private var modeBatch = false
   120→    private var began = false
   121→
   122→    private fun setMode(batchMode: Boolean) {
   123→        if (batchMode == modeBatch && began) return
   124→        if (began) {
   125→            if (modeBatch) batch.end() else sr.end()
   126→        }
   127→        modeBatch = batchMode
   128→        if (modeBatch) {
   129→            batch.begin()
   130→        } else {
   131→            sr.begin(ShapeRenderer.ShapeType.Filled)
   132→        }
   133→        began = true
   134→    }
   135→
   136→    fun fogged(out: Color, base: Color, z: Float): Color {
   137→        val f = proj.fog(z) * 0.85f
   138→        out.set(base)
   139→        out.r += (Palette.FOG.r - out.r) * f
   140→        out.g += (Palette.FOG.g - out.g) * f
   141→        out.b += (Palette.FOG.b - out.b) * f
   142→        return out
   143→    }

   145→    // ── 3D box helper (matches WorldRenderer.box3D) ────────────────────
   146→    private fun box3D(
   147→        wx: Float, w: Float, h: Float, zFront: Float, zBack: Float,
   148→        front: Color, side: Color, top: Color, sx: Float, sy: Float
   149→    ) {
   150→        val sF = proj.scale(zFront)
   151→        val sB = proj.scale(zBack)
   152→        val xlF = proj.screenX(wx - w / 2f, zFront) + sx
   153→        val xrF = proj.screenX(wx + w / 2f, zFront) + sx
   154→        val ybF = proj.groundY(zFront) + sy
   155→        val ytF = ybF - h * proj.ppu * sF
   156→        val xlB = proj.screenX(wx - w / 2f, zBack) + sx
   157→        val xrB = proj.screenX(wx + w / 2f, zBack) + sx
   158→        val ybB = proj.groundY(zBack) + sy
   159→        val ytB = ybB - h * proj.ppu * sB
   160→
   161→        sr.setColor(top)
   162→        sr.triangle(xlF, ytF, xrF, ytF, xrB, ytB)
   163→        sr.triangle(xlF, ytF, xrB, ytB, xlB, ytB)
   164→        val camWx = proj.camX
   165→        if (camWx < wx - w / 2f) {
   166→            sr.setColor(side)
   167→            sr.triangle(xlF, ytF, xlB, ytB, xlB, ybB)
   168→            sr.triangle(xlF, ytF, xlB, ybB, xlF, ybF)
   169→        } else if (camWx > wx + w / 2f) {
   170→            sr.setColor(side)
   171→            sr.triangle(xrF, ytF, xrB, ytB, xrB, ybB)
   172→            sr.triangle(xrF, ytF, xrB, ybB, xrF, ybF)
   173→        }
   174→        sr.setColor(front)
   175→        sr.rect(xlF, ytF, xrF - xlF, ybF - ytF)
   176→    }

   179→    private fun train(t: Train, sx: Float, sy: Float) {
   180→        val livery = Palette.TRAIN_LIVERIES[t.livery % Palette.TRAIN_LIVERIES.size]
   181→        val base = tmpC2.set(Color(livery[0]))
   182→        val carLen = GameConfig.TRAIN_CAR_LENGTH
   183→        val w = GameConfig.TRAIN_WIDTH
   184→        val h = GameConfig.TRAIN_HEIGHT
   185→
   186→        for (car in t.cars - 1 downTo 0) {
   187→            val zF = t.z - car * carLen
   188→            val zB = zF - carLen + 0.55f // coupling gap
   189→            if (zB > GameConfig.VIEW_DISTANCE + 4f || zF < -10f) continue
   190→            val zFrontC = zF.coerceAtLeast(-9f)
   191→            fogged(tmpC, base, zF)
   192→            val front = Color(tmpC)
   193→            val sideC = tmpC.cpy().mul(0.68f)
   194→            val topC = tmpC.cpy().mul(1.18f)
   195→
   196→            // undercarriage shadow
   197→            val sF = proj.scale(zFrontC)
   198→            sr.setColor(0f, 0f, 0f, 0.3f)
   199→            val shW = w * proj.ppu * sF * 0.6f
   200→            for (lane in t.lanes) {
   201→                val lx = proj.screenX(lane * GameConfig.LANE_WIDTH.toFloat(), zFrontC) + sx
   202→                sr.rect(lx - shW / 2, proj.groundY(zFrontC) + sy - 2f, shW, 5f * sF + 2f)
   203→            }
   204→
   205→            for (lane in t.lanes) {
   206→                val wx = lane * GameConfig.LANE_WIDTH.toFloat()
   207→                box3D(wx, w, h, zFrontC, zB.coerceAtLeast(-9f), front, sideC, topC, sx, sy)
   208→                trainDetails(t, wx, zFrontC, zB, w, h, livery, sx, sy)
   209→            }
   210→        }
   211→    }

   213→    private fun trainDetails(t: Train, wx: Float, zF: Float, zB: Float, w: Float, h: Float, livery: IntArray, sx: Float, sy: Float) {
   214→        if (zF > GameConfig.VIEW_DISTANCE * 0.92f || zF < -8f) return
   215→        val s = proj.scale(zF)
   216→        val xl = proj.screenX(wx - w / 2f, zF) + sx
   217→        val xr = proj.screenX(wx + w / 2f, zF) + sx
   218→        val yb = proj.groundY(zF) + sy
   219→        val yt = yb - h * proj.ppu * s
   220→
   221→        // stripe along the bottom
   222→        sr.setColor(fogged(tmpC, Color(livery[1]), zF))
   223→        sr.rect(xl, yb - h * 0.22f * proj.ppu * s, xr - xl, h * 0.1f * proj.ppu * s)
   224→
   225→        // windshield / front face details only on front-facing car (zF close to t.z)
   226→        val isFront = abs(zF - t.z) < 0.1f
   227→        if (isFront) {
   228→            // windshield
   229→            sr.setColor(fogged(tmpC, Color(0x1f2a30ff.toInt()), zF))
   230→            sr.rect(xl + (xr - xl) * 0.15f, yb - h * 0.82f * proj.ppu * s, (xr - xl) * 0.7f, h * 0.3f * proj.ppu * s)
   231→            // headlights
   232→            val glowCol = if (t.kind == 2) 1f else 0.75f
   233→            sr.setColor(fogged(tmpC, Color(1f, 0.95f, 0.7f, 1f), zF))
   234→            sr.circle(xl + (xr - xl) * 0.16f, yb - h * 0.3f * proj.ppu * s, 4.5f * s)
   235→            sr.circle(xl + (xr - xl) * 0.84f, yb - h * 0.3f * proj.ppu * s, 4.5f * s)
   236→            if (t.kind == 2) {
   237→                // approaching: headlight beam glow
   238→                sr.setColor(1f, 0.95f, 0.75f, 0.16f)
   239→                sr.circle(xl + (xr - xl) * 0.5f, yb - h * 0.4f * proj.ppu * s, (xr - xl) * 0.75f)
   240→            }
   241→            // number plate
   242→            sr.setColor(fogged(tmpC, Color(0xfff2dcff.toInt()), zF))
   243→            sr.rect(xl + (xr - xl) * 0.42f, yb - h * 0.5f * proj.ppu * s, (xr - xl) * 0.16f, h * 0.09f * proj.ppu * s)
   244→        } else {
   245→            // side windows (visible on side face)
   246→            val camWx = proj.camX
   247→            val leftSide = camWx < wx - w / 2f
   248→            val sxE = proj.screenX(if (leftSide) wx - w / 2f else wx + w / 2f, zF) + sx
   249→            val sxE2 = proj.screenX(if (leftSide) wx - w / 2f else wx + w / 2f, zB) + sx
   250→            val ybE = proj.groundY(zF) + sy
   251→            val ytE = ybE - h * proj.ppu * s
   252→            sr.setColor(fogged(tmpC, Color(0x24343bff.toInt()), zF))
   253→            val winN = 4
   254→            for (i in 0 until winN) {
   255→                val ft0 = (i + 0.15f) / winN
   256→                val ft1 = (i + 0.85f) / winN
   257→                val wx0 = sxE + (sxE2 - sxE) * ft0
   258→                val wx1 = sxE + (sxE2 - sxE) * ft1
   259→                val wy0 = ybE - h * 0.78f * proj.ppu * s + (proj.groundY(zB) + sy - h * proj.ppu * proj.scale(zB) - (ybE - h * 0.78f * proj.ppu * s)) * ft0
   260→                sr.rect(wx0.coerceAtLeast(wx1), wy0, abs(wx1 - wx0), h * 0.24f * proj.ppu * s)
   261→            }
   262→            // door
   263→            sr.setColor(fogged(tmpC, Color(livery[1]).mul(0.9f), zF))
   264→            val dx = sxE + (sxE2 - sxE) * 0.02f
   265→            sr.rect(dx.coerceAtLeast(sxE2 * 0.99f + sxE * 0.01f), ybE - h * 0.62f * proj.ppu * s, abs(sxE2 - sxE) * 0.12f, h * 0.5f * proj.ppu * s)
   266→        }
   267→
   268→        // graffiti tag (original art): colored arcs seeded per train
   269→        if (t.seed % 3 == 0 && !isFront) {
   270→            val rng = Random(t.seed + (zF * 10).toInt())
   271→            val cols = arrayOf(Color(0xffd24aff.toInt()), Color(0x2dd4bfff.toInt()), Color(0xe2493bff.toInt()))
   272→            sr.setColor(fogged(tmpC, cols[rng.nextInt(cols.size)], zF))
   273→            val gy = yb - h * (0.3f + rng.nextFloat() * 0.2f) * proj.ppu * s
   274→            sr.rect(xl + (xr - xl) * 0.1f, gy, (xr - xl) * (0.2f + rng.nextFloat() * 0.3f), 4f * s + 1f)
   275→        }
   276→    }

   279→    private fun obstacle(o: Obstacle, sx: Float, sy: Float) {
   280→        val s = proj.scale(o.z)
   281→        if (o.z > GameConfig.VIEW_DISTANCE || o.z < -8f) return
   282→        when (o.kind) {
   283→            ObstacleKind.LOW_BARRIER -> lowBarrier(o, sx, sy, s)
   284→            ObstacleKind.HIGH_BARRIER -> highBarrier(o, sx, sy, s)
   285→            ObstacleKind.GATE -> gate(sx, sy, s)
   286→            ObstacleKind.FENCE_FULL -> fenceFull(sx, sy, s)
   287→            ObstacleKind.BLOCKADE -> blockade(o, sx, sy, s)
   288→        }
   289→    }

   291→    private fun stripeBoard(xl: Float, xr: Float, y: Float, hpx: Float, variant: Int) {
   292→        // orange/white diagonal hazard stripes
   293→        val n = 6
   294→        val w = xr - xl
   295→        for (i in 0 until n) {
   296→            val segW = w / n
   297→            val even = (i + variant) % 2 == 0
   298→            sr.setColor(if (even) Color(0xf28c3cff.toInt()) else Color(0xfff2dcff.toInt()))
   299→            sr.rect(xl + i * segW, y, segW + 0.5f, hpx)
   300→        }
   301→    }

   303→    private fun lowBarrier(o: Obstacle, sx: Float, sy: Float, s: Float) {
   304→        val wx = o.lane * GameConfig.LANE_WIDTH.toFloat()
   305→        val xl = proj.screenX(wx - 1.0f, o.z) + sx
   306→        val xr = proj.screenX(wx + 1.0f, o.z) + sx
   307→        val yb = proj.groundY(o.z) + sy
   308→        val hpx = 0.95f * proj.ppu * s
   309→        // legs
   310→        sr.setColor(fogged(tmpC, Color(0x555049ff.toInt()), o.z))
   311→        sr.rect(xl + (xr - xl) * 0.08f, yb - hpx, 5f * s, hpx * 0.55f)
   312→        sr.rect(xr - (xr - xl) * 0.08f - 5f * s, yb - hpx, 5f * s, hpx * 0.55f)
   313→        // striped board
   314→        stripeBoard(xl, xr, yb - hpx, hpx * 0.45f, o.variant)
   315→        sr.setColor(fogged(tmpC, Color(0x2b2b2bff.toInt()), o.z))
   316→        sr.rect(xl, yb - hpx, xr - xl, 2.2f * s)
   317→    }

   319→    private fun highBarrier(o: Obstacle, sx: Float, sy: Float, s: Float) {
   320→        val wx = o.lane * GameConfig.LANE_WIDTH.toFloat()
   321→        val xl = proj.screenX(wx - 1.1f, o.z) + sx
   322→        val xr = proj.screenX(wx + 1.1f, o.z) + sx
   323→        val yb = proj.groundY(o.z) + sy
   324→        // supports from ground to 1.15u (slide gap below)
   325→        sr.setColor(fogged(tmpC, Color(0x555049ff.toInt()), o.z))
   326→        sr.rect(xl, yb - 1.2f * proj.ppu * s, 6f * s, 1.2f * proj.ppu * s)
   327→        sr.rect(xr - 6f * s, yb - 1.2f * proj.ppu * s, 6f * s, 1.2f * proj.ppu * s)
   328→        // overhead signboard 1.2u..2.2u — teal = slide action (cool color)
   329→        val boardY = yb - 2.3f * proj.ppu * s
   330→        val boardH = 1.05f * proj.ppu * s
   331→        sr.setColor(fogged(tmpC, Color(0x2fa08bff.toInt()), o.z))
   332→        sr.rect(xl, boardY, xr - xl, boardH)
   333→        sr.setColor(fogged(tmpC, Color(0x8ff2e2ff.toInt()), o.z))
   334→        // down-arrow marks (slide cue)
   335→        val cx = (xl + xr) / 2
   336→        val n = 3
   337→        for (i in 0 until n) {
   338→            val ax = xl + (xr - xl) * (0.3f + i * 0.2f)
   339→            sr.rect(ax - 2.5f * s, boardY + boardH * 0.3f, 5f * s, boardH * 0.4f)
   340→            sr.rect(ax - 6f * s, boardY + boardH * 0.22f, 12f * s, 3.5f * s)
   341→        }
   342→        sr.setColor(0f, 0f, 0f, 0.25f)
   343→        sr.rect(xl, boardY, xr - xl, 2.5f * s)
   344→    }

   346→    private fun gate(sx: Float, sy: Float, s: Float) {
   347→        val xl = proj.screenX(-GameConfig.LANE_WIDTH * 1.5f - 0.4f, GameConfig.MIN_PATTERN_GAP.coerceAtMost(10f)) + sx
   348→        val z = proj.camX // unused guard
   349→        val zNow = proj.groundY(0f)
   350→        val xL = proj.screenX(-GameConfig.LANE_WIDTH * 1.5f - 0.5f, 10f) + sx
   351→        val xR = proj.screenX(GameConfig.LANE_WIDTH * 1.5f + 0.5f, 10f) + sx
   352→        val yb = proj.groundY(10f) + sy
   353→        // posts
   354→        sr.setColor(fogged(tmpC, Color(0x555049ff.toInt()), 10f))
   355→        sr.rect(xL - 7f * s, yb - 2.4f * proj.ppu * proj.scale(10f), 14f * s, 2.4f * proj.ppu * proj.scale(10f))
   356→        sr.rect(xR - 7f * s, yb - 2.4f * proj.ppu * proj.scale(10f), 14f * s, 2.4f * proj.ppu * proj.scale(10f))
   357→        val boardY = yb - 2.35f * proj.ppu * proj.scale(10f)
   358→        val boardH = 1.1f * proj.ppu * proj.scale(10f)
   359→        sr.setColor(fogged(tmpC, Color(0x2fa08bff.toInt()), 10f))
   360→        sr.rect(xL - 7f * s, boardY, (xR - xL) + 14f * s, boardH)
   361→        sr.setColor(fogged(tmpC, Color(0x8ff2e2ff.toInt()), 10f))
   362→        var i = 0
   363→        val n = 8
   364→        while (i < n) {
   365→            val ax = xL + (xR - xL) * (0.08f + i * 0.12f)
   366→            sr.rect(ax - 2.2f * s, boardY + boardH * 0.28f, 4.4f * s, boardH * 0.42f)
   367→            sr.rect(ax - 5.5f * s, boardY + boardH * 0.2f, 11f * s, 3f * s)
   368→            i++
   369→        }
   370→    }

   372→    private fun fenceFull(sx: Float, sy: Float, s: Float) {
   373→        val xL = proj.screenX(-GameConfig.LANE_WIDTH * 1.5f - 0.5f, 10f) + sx
   374→        val xR = proj.screenX(GameConfig.LANE_WIDTH * 1.5f + 0.5f, 10f) + sx
   375→        val yb = proj.groundY(10f) + sy
   376→        val hpx = 0.95f * proj.ppu * proj.scale(10f)
   377→        sr.setColor(fogged(tmpC, Color(0x555049ff.toInt()), 10f))
   378→        sr.rect(xL, yb - hpx, 6f * s, hpx)
   379→        sr.rect(xR - 6f * s, yb - hpx, 6f * s, hpx)

   401→    private fun coin(c: Coin, sx: Float, sy: Float) {
   402→        if (c.z > GameConfig.VIEW_DISTANCE || c.z < -5f) return

   406→        c.phase += 0.16f
   407→        val frame = ((c.phase % 1f) * 10f).toInt() % 10
   408→        val bob = sin(c.phase * 6.28f) * 2.5f * s
   409→        val size = 30f * s
   410→        // glow
   411→        batch.setColor(1f, 0.85f, 0.35f, 0.35f * s.coerceIn(0f, 1f))
   412→        batch.draw(TextureGen.glow, x - size * 0.95f, y - size * 0.95f + bob, size * 1.9f, size * 1.9f)
   413→        batch.setColor(1f, 1f, 1f, 1f)
   414→        batch.draw(TextureGen.coinFrames[frame], x - size / 2, y - size / 2 + bob, size, size)
   415→    }

   417→    private fun powerup(p: PowerUpPickup, sx: Float, sy: Float) {
   418→        if (p.z > GameConfig.VIEW_DISTANCE || p.z < -5f) return
   419→        val s = proj.scale(p.z)

   423→        val y = proj.groundY(p.z) + sy - (1.35f * proj.ppu * s) + bob
   424→        val size = 52f * s

   426→        // pulsing ring + glow + icon
   427→        batch.setColor(color.r, color.g, color.b, 0.5f)
   428→        batch.draw(TextureGen.glow, x - size, y - size + size / 2, size * 2, size * 2)
   429→        batch.setColor(1f, 1f, 1f, 0.95f)
   430→        TextureGen.panelNine.draw(batch, x - size / 2, y - size / 2 + size / 2, size, size)
   431→        batch.setColor(1f, 1f, 1f, 1f)
   432→        batch.draw(TextureGen.powerIcons[p.type], x - size * 0.34f, y - size * 0.34f + size / 2, size * 0.68f, size * 0.68f)

   445→    // ── Player (procedural runner, seen from behind) ───────────────────
   446→    private fun player(p: Player, ch: CharacterDef, sx: Float, sy: Float, blink: Boolean, shieldOn: Boolean, boostOn: Boolean) {
   447→        val s = 1f
   448→        val x = proj.screenX(p.x, 0f) + sx
   449→        val groundY = proj.groundY(0f) + sy
   450→        val u = proj.ppu // pixels per unit at z=0
   451→
   452→        // shadow
   453→        val shadowScale = Mathz.clamp01(1f - p.jumpY / 5f)
   454→        sr.setColor(0f, 0f, 0f, 0.35f * shadowScale)
   455→        sr.ellipse(x - u * 0.42f * shadowScale, groundY - 4f, u * 0.84f * shadowScale, u * 0.16f * shadowScale + 2f)
   456→
   457→        if (blink && (p.invulnTimer > 0f) && ((p.invulnTimer * 10f).toInt() % 2 == 0)) return
   458→
   459→        val bodyH = u * GameConfig.PLAYER_HEIGHT
   460→        val squash = 1f - p.squash * 0.16f
   461→        val isJump = p.state == PlayerState.JUMPING
   462→        val isSlide = p.state == PlayerState.SLIDING
   463→        val isDead = p.state == PlayerState.DEAD
   464→
   465→        val cx = x
   466→        val by = groundY - p.jumpY * u
   467→
   468→        if (isSlide) {
   469→            drawSlidePose(p, ch, cx, by, u)
   470→            return
   471→        }
   472→
   473→        sr.setColor(Color.WHITE)
   474→        val rotate = if (isDead) p.deathSpin else p.lean * 0.5f
   475→        // all shapes around a pivot at feet; use manual rotation via cos/sin offsets
   476→        val rotRad = rotate * PI.toFloat() / 180f
   477→        fun rot(px: Float, py: Float, ox: Float, oy: Float): Pair<Float, Float> {
   478→            val dx = px - ox; val dy = py - oy
   479→            return (ox + dx * cos(rotRad) - dy * sin(rotRad)) to (oy + dx * sin(rotRad) + dy * cos(rotRad))
   480→        }
   481→
   482→        val swing = if (isJump) 0f else sin(p.runPhase)
   483→        val swing2 = if (isJump) 0f else sin(p.runPhase + PI.toFloat())
   484→        val hipY = by + bodyH * 0.42f * squash
   485→        val shoulderY = by + bodyH * 0.78f * squash
   486→        val headY = by + bodyH * 0.92f * squash
   487→
   488→        // legs (runner shorts + sneakers)
   489→        val legColor = tmpC.set(Color(ch.pants))
   490→        sr.setColor(legColor)
   491→        val footLY = by + max(0f, swing) * bodyH * 0.12f
   492→        val footRY = by + max(0f, swing2) * bodyH * 0.12f
   493→        // left leg
   494→        sr.rect(cx - u * 0.16f, footLY, u * 0.13f, hipY - footLY)
   495→        sr.rect(cx + u * 0.04f, footRY, u * 0.13f, hipY - footRY)
   496→        // sneakers (accent color)
   497→        sr.setColor(Color(ch.shoes))
   498→        sr.rect(cx - u * 0.19f, footLY - u * 0.05f, u * 0.19f, u * 0.08f)
   499→        sr.rect(cx + u * 0.02f, footRY - u * 0.05f, u * 0.19f, u * 0.08f)
   500→
   501→        // backpack (seen from behind)
   502→        sr.setColor(Color(ch.backpack))
   503→        val bw = u * 0.42f
   504→        sr.rect(cx - bw / 2, hipY - u * 0.02f, bw, shoulderY - hipY + u * 0.1f)
   505→        sr.setColor(Color(ch.accent))
   506→        sr.rect(cx - bw / 2 + u * 0.06f, shoulderY - u * 0.16f, bw - u * 0.12f, u * 0.1f)
   507→
   508→        // torso hoodie
   509→        sr.setColor(Color(ch.hoodie))
   510→        sr.rect(cx - u * 0.26f, hipY, u * 0.52f, shoulderY - hipY + u * 0.04f)
   511→        // hood bump
   512→        sr.circle(cx, shoulderY + u * 0.02f, u * 0.1f)
   513→
   514→        // arms pumping
   515→        val armColor = tmpC.set(Color(ch.hoodie)).mul(0.85f)
   516→        sr.setColor(armColor)
   517→        val armSwing = if (isJump) -0.3f else swing * 0.5f
   518→        val armSwing2 = if (isJump) -0.3f else swing2 * 0.5f
   519→        sr.rect(cx - u * 0.38f, shoulderY - u * 0.1f + armSwing * u * 0.16f, u * 0.11f, bodyH * 0.26f)
   520→        sr.rect(cx + u * 0.27f, shoulderY - u * 0.1f + armSwing2 * u * 0.16f, u * 0.11f, bodyH * 0.26f)
   521→        // hands
   522→        sr.setColor(Color(ch.skin))
   523→        sr.circle(cx - u * 0.325f, shoulderY - u * 0.12f + armSwing * u * 0.16f, u * 0.055f)
   524→        sr.circle(cx + u * 0.325f, shoulderY - u * 0.12f + armSwing2 * u * 0.16f, u * 0.055f)
   525→
   526→        // head + backward cap
   527→        sr.setColor(Color(ch.skin))
   528→        sr.circle(cx, headY, u * 0.17f)
   529→        sr.setColor(Color(ch.cap))
   530→        sr.circle(cx, headY + u * 0.06f, u * 0.17f)
   531→        sr.rect(cx - u * 0.17f, headY + u * 0.03f, u * 0.34f, u * 0.07f)
   532→        sr.setColor(Color(ch.accent))
   533→        sr.rect(cx - u * 0.06f, headY + u * 0.16f, u * 0.12f, u * 0.05f)
   534→
   535→        // shield bubble (double circle = rim + glass)
   536→        if (shieldOn) {
   537→            sr.setColor(0.5f, 1f, 0.95f, 0.6f)
   538→            sr.circle(cx, by + bodyH * 0.5f, bodyH * 0.62f)
   539→            sr.setColor(0.3f, 0.9f, 0.85f, 0.22f)
   540→            sr.circle(cx, by + bodyH * 0.5f, bodyH * 0.58f)
   541→        }
   542→    }

   615→    private fun drawFloatingTexts(p: Particles, sx: Float, sy: Float) {

   625→        fontLarge.setColor(1f, 1f, 1f, 1f)
